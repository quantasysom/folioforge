cmake_minimum_required(VERSION 3.25)
if(POLICY CMP0144)
  cmake_policy(SET CMP0144 NEW)
endif()
project(FolioForge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
option(FOLIOFORGE_DESKTOP "Build the Qt desktop application" ON)
find_package(qpdf 12 CONFIG REQUIRED)
find_package(PDFium REQUIRED)
add_library(pdfengine engine/src/document.cpp engine/src/file_io.cpp engine/src/text_edit.cpp engine/src/font_codec.cpp engine/src/sfnt.cpp engine/src/annotation.cpp engine/src/form.cpp engine/src/signing.cpp engine/src/navigation.cpp)
if(APPLE)
  target_link_libraries(pdfengine PRIVATE "-framework Security" "-framework CoreFoundation")
endif()
target_include_directories(pdfengine PUBLIC engine/include)
target_link_libraries(pdfengine PRIVATE qpdf::libqpdf)
add_library(pdfengine-pdfium engine/src/renderer.cpp engine/src/render_service.cpp)
if(WIN32)
  target_sources(pdfengine-pdfium PRIVATE engine/src/process_win.cpp)
else()
  target_sources(pdfengine-pdfium PRIVATE engine/src/process_posix.cpp)
endif()
target_include_directories(pdfengine-pdfium PRIVATE engine/src)
target_include_directories(pdfengine-pdfium PUBLIC engine/include)
target_link_libraries(pdfengine-pdfium PUBLIC pdfengine PRIVATE PDFium::PDFium)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
  target_compile_options(pdfengine PRIVATE /W4 /permissive- /utf-8)
  target_compile_definitions(pdfengine PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_definitions(pdfengine-pdfium PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_executable(pdfeditor-render apps/render/main.cpp)
target_include_directories(pdfeditor-render PRIVATE engine/src)
target_link_libraries(pdfeditor-render PRIVATE pdfengine-pdfium)
add_executable(pdfeditor-cli apps/cli/main.cpp)
target_link_libraries(pdfeditor-cli PRIVATE pdfengine-pdfium)
add_dependencies(pdfeditor-cli pdfeditor-render)
if(FOLIOFORGE_DESKTOP)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Concurrent)
  qt_add_executable(pdfeditor-desktop WIN32 apps/desktop/main.cpp apps/desktop/window.cpp)
  target_link_libraries(pdfeditor-desktop PRIVATE pdfengine-pdfium Qt6::Widgets Qt6::Concurrent)
  add_dependencies(pdfeditor-desktop pdfeditor-render)
endif()
if(BUILD_TESTING)
  # Let tests find PDFium/QPDF/Qt shared libraries without manual PATH/LD_LIBRARY_PATH setup.
  set(runtime_dirs "${PDFIUM_RUNTIME_DIR}" "$<TARGET_FILE_DIR:qpdf::libqpdf>")
  if(FOLIOFORGE_DESKTOP AND Qt6_DIR)
    get_filename_component(qt_prefix "${Qt6_DIR}/../../.." ABSOLUTE)
    list(APPEND runtime_dirs "${qt_prefix}/bin" "${qt_prefix}/lib")
  endif()
  add_executable(engine-tests tests/engine_tests.cpp)
  target_link_libraries(engine-tests PRIVATE pdfengine-pdfium qpdf::libqpdf)
  add_test(NAME engine-workflows COMMAND engine-tests)
  add_executable(text-edit-tests tests/text_edit_tests.cpp)
  target_link_libraries(text-edit-tests PRIVATE pdfengine-pdfium qpdf::libqpdf)
  add_test(NAME text-edit-workflows COMMAND text-edit-tests)
  add_executable(annotation-tests tests/annotation_tests.cpp)
  target_link_libraries(annotation-tests PRIVATE pdfengine-pdfium qpdf::libqpdf)
  add_test(NAME annotation-workflows COMMAND annotation-tests)
  add_executable(form-tests tests/form_tests.cpp)
  target_link_libraries(form-tests PRIVATE pdfengine-pdfium qpdf::libqpdf)
  add_test(NAME form-workflows COMMAND form-tests)
  add_executable(navigation-tests tests/navigation_tests.cpp)
  target_link_libraries(navigation-tests PRIVATE pdfengine-pdfium qpdf::libqpdf)
  add_test(NAME navigation-workflows COMMAND navigation-tests)
  add_executable(redaction-tests tests/redaction_tests.cpp)
  target_link_libraries(redaction-tests PRIVATE pdfengine-pdfium qpdf::libqpdf)
  add_test(NAME redaction-workflows COMMAND redaction-tests)
  add_executable(signing-tests tests/signing_tests.cpp)
  target_link_libraries(signing-tests PRIVATE pdfengine-pdfium qpdf::libqpdf)
  target_compile_definitions(signing-tests PRIVATE FOLIOFORGE_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
  add_test(NAME signing-workflows COMMAND signing-tests)
  set_tests_properties(signing-workflows PROPERTIES SKIP_RETURN_CODE 77)
  add_executable(font-tests tests/font_tests.cpp)
  target_link_libraries(font-tests PRIVATE pdfengine-pdfium qpdf::libqpdf)
  target_include_directories(font-tests PRIVATE engine/src)
  add_test(NAME font-workflows COMMAND font-tests)
  set_tests_properties(font-workflows PROPERTIES SKIP_RETURN_CODE 77)
  add_executable(render-service-tests tests/render_service_tests.cpp)
  target_link_libraries(render-service-tests PRIVATE pdfengine-pdfium)
  add_dependencies(render-service-tests pdfeditor-render)
  add_test(NAME render-service-workflows COMMAND render-service-tests "$<TARGET_FILE:pdfeditor-render>")
  add_executable(history-tests tests/history_tests.cpp)
  target_link_libraries(history-tests PRIVATE pdfengine-pdfium)
  add_test(NAME history-workflows COMMAND history-tests)
  add_executable(image-tests tests/image_tests.cpp)
  target_link_libraries(image-tests PRIVATE pdfengine-pdfium)
  add_test(NAME image-workflows COMMAND image-tests)
  set_tests_properties(engine-workflows PROPERTIES FIXTURES_SETUP pdf-corpus)
  if(FOLIOFORGE_DESKTOP)
    if(WIN32)
      find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell REQUIRED)
      add_test(NAME desktop-smoke COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/desktop-smoke.ps1"
        -Executable "$<TARGET_FILE:pdfeditor-desktop>" -WorkingDirectory "${CMAKE_CURRENT_BINARY_DIR}")
    else()
      add_test(NAME desktop-smoke COMMAND pdfeditor-desktop --smoke-test)
    endif()
    set_tests_properties(desktop-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45 FIXTURES_REQUIRED pdf-corpus)
  endif()
endif()
option(FOLIOFORGE_PACKAGING "Enable runtime deployment and CPack installers" OFF)
if(FOLIOFORGE_PACKAGING)
  include(cmake/Packaging.cmake)
endif()
if(BUILD_TESTING)
  get_property(all_tests DIRECTORY PROPERTY TESTS)
  foreach(test IN LISTS all_tests)
    foreach(dir IN LISTS runtime_dirs)
      foreach(variable PATH LD_LIBRARY_PATH DYLD_LIBRARY_PATH)
        set_property(TEST ${test} APPEND PROPERTY ENVIRONMENT_MODIFICATION "${variable}=path_list_prepend:${dir}")
      endforeach()
    endforeach()
  endforeach()
endif()
